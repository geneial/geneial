file(GLOB TEST_SRC CONFIGURE_DEPENDS suites/*_test.cpp)
foreach(test_source ${TEST_SRC})
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE geneial::geneial geneial_options)
    if(GENEIAL_HEADER_ONLY_TESTS)
        target_link_libraries(${test_name} PRIVATE Boost::headers)
        target_compile_definitions(${test_name} PRIVATE GENEIAL_HEADER_ONLY_TESTS)
    else()
        target_link_libraries(${test_name} PRIVATE Boost::unit_test_framework)
    endif()
    if(test_name STREQUAL "art_test")
        target_compile_definitions(${test_name} PRIVATE
            GENEIAL_ART_TARGET="${CMAKE_CURRENT_SOURCE_DIR}/../../demo/showcase/assets/mona_lisa.ppm")
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 120)
endforeach()
