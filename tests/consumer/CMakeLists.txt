cmake_minimum_required(VERSION 3.21)
project(geneial_consumer LANGUAGES CXX)
find_package(geneial 0.3 CONFIG REQUIRED)
enable_testing()
foreach(kind geneial geneial_shared)
    add_executable(consumer_${kind} main.cpp)
    target_link_libraries(consumer_${kind} PRIVATE geneial::${kind})
    add_test(NAME consumer_${kind} COMMAND consumer_${kind})
endforeach()
